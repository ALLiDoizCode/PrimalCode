**Test Structure:**
```typescript
describe('EcosystemObserverTool', () => {
  let tool: EcosystemObserverTool;
  let mockEcosystem: jest.Mocked<MockEcosystemState>;
  
  beforeEach(() => {
    mockEcosystem = createMockEcosystemState();
    tool = new EcosystemObserverTool(mockEcosystem);
  });
  
  it('should generate engaging ecosystem description', async () => {
    const result = await tool.execute({ route_id: 'forest_path' });
    expect(result.currentState).toContain('forest path');
    expect(result.monsterBehaviors).toBeInstanceOf(Array);
    expect(result.suggestedActions).toBeInstanceOf(Array);
  });
});
```
