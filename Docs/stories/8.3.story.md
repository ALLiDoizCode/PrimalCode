### Enhanced WebSocket Client Integration Pattern
```typescript
// Extended WebSocket client for debug interfaces
import { useWebSocket } from "@/lib/dashboard/websocket-client"

interface DebugDataHook {
  data: any
  connected: boolean
  error: string | null
  historicalData: any[]
}

export function useDebugWebSocket(interfaceType: string, processId: string): DebugDataHook {
  const { data, connected, error } = useWebSocket(`ws://localhost:8080/debug/${interfaceType}/${processId}`)
  const [historicalData, setHistoricalData] = useState<any[]>([])
  
  useEffect(() => {
    if (data) {
      setHistoricalData(prev => [...prev.slice(-100), data]) // Keep last 100 entries
    }
  }, [data])
  
  return { data, connected, error, historicalData }
}
```
