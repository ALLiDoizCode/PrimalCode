### Test Example Pattern
[Source: architecture/testing-strategy.md#mcp-tool-test]
```typescript
describe('PlaceFoodTool', () => {
  let tool: PlaceFoodTool;
  let mockMonsterSystem: jest.Mocked<MockMonsterSystem>;
  let mockEnvironmentState: jest.Mocked<MockEnvironmentState>;
  
  beforeEach(() => {
    mockMonsterSystem = createMockMonsterSystem();
    mockEnvironmentState = createMockEnvironmentState();
    tool = new PlaceFoodTool(mockMonsterSystem, mockEnvironmentState);
  });
  
  it('should place food and trigger monster behavioral adaptation', async () => {
    // Arrange
    const mockEnvironment = createMockEnvironment();
    const mockMonsters = createMockMonsters();
    mockEnvironmentState.getEnvironmentState.mockResolvedValue(mockEnvironment);
    mockMonsterSystem.getMonstersInRoute.mockResolvedValue(mockMonsters);
    
    // Act
    const result = await tool.execute({ 
      route_id: 'forest_path', 
      location: { x: 100, y: 100 }, 
      food_type: 'meat' 
    });
    
    // Assert
    expect(result.placementConfirmation).toContain('food source placed');
    expect(result.behavioralPredictions).toContain('monsters will be attracted');
    expect(mockEnvironmentState.addFoodSource).toHaveBeenCalledWith('forest_path', { x: 100, y: 100 }, 'meat');
    expect(mockMonsterSystem.notifyEnvironmentalChange).toHaveBeenCalled();
  });
});
```
